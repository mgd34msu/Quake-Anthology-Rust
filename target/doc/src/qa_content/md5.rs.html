1//! Doom 3 / rerelease skeletal models (MD5 v10 mesh and animation).
2//!
3//! Donor provenance: `src/formats/q3-model/md5.ts` (mesh, animation,
4//! scales, and skinning from the Q1/Q2 rerelease ports and q2repro
5//! `refresh/models.c`, `mesh.c`), with tokens from [`crate::model_text`]
6//! and quaternions from [`crate::quaternion`]. Scale files are JSON; the
7//! small strict parser below preserves the donor fallback to unit scales
8//! with diagnostics.
9//!
10//! Single-`fround` expressions compute in `f64` and cast once at the
11//! store, per the `qa-core` math discipline; per-operation roundings reuse
12//! [`qa_core::math`].

229fn md5_header(tokens: &mut ModelTokens) -> Result<String, BinaryError> {
230    tokens.expect("MD5Version")?;
231    tokens.expect("10")?;
232    tokens.expect("commandline")?;
233    tokens.token()
234}

236fn read_mesh(tokens: &mut ModelTokens, joints: &[SkeletonJointPose]) -> Result<Md5Mesh, BinaryError> {
237    tokens.expect("mesh")?;
238    tokens.expect("{")?;
239    tokens.expect("shader")?;
240    let shader = tokens.token()?;
241    tokens.expect("numverts")?;
242    let vertex_count = tokens.integer(0, 65535)? as usize;
243    let mut vertices = indexed_records(tokens, vertex_count, "vert", |tokens| {
244        tokens.expect("(")?;
245        let tex_coord = qa_core::math::vec2(tokens.float()?, tokens.float()?);
246        tokens.expect(")")?;
247        Ok(Md5Vertex {
248            tex_coord,
249            normal: vec3(0.0, 0.0, 0.0),
250            weights: IndexRange {
251                first: tokens.integer(0, i32::MAX)? as u32,
252                count: tokens.integer(0, i32::MAX)? as u32,
253            },
254        })
255    })?;
256    tokens.expect("numtris")?;
257    let triangle_count = tokens.integer(0, 65535)? as usize;
258    let triangles = indexed_records(tokens, triangle_count, "tri", |tokens| {
259        Ok([
260            tokens.integer(0, vertex_count as i32 - 1)? as u32,
261            tokens.integer(0, vertex_count as i32 - 1)? as u32,
262            tokens.integer(0, vertex_count as i32 - 1)? as u32,
263        ])
264    })?;
265    tokens.expect("numweights")?;
266    let weight_count = tokens.integer(0, 1_048_576)? as usize;
267    let weights = indexed_records(tokens, weight_count, "weight", |tokens| {
268        let joint = tokens.integer(0, joints.len() as i32 - 1)? as u32;
269        let bias = tokens.float()?;
270        if !(0.0..=1.0).contains(&bias) {
271            return tokens.fail(format!("weight bias {bias} outside 0..1"));
272        }
273        Ok(Md5Weight {
274            joint,
275            bias,
276            position: tokens.vector()?,
277        })
278    })?;
279    tokens.expect("}")?;
280    for vertex in &vertices {
281        if vertex.weights.count as usize > weight_count
282            || vertex.weights.first as usize > weight_count - vertex.weights.count as usize
283        {
284            return tokens.fail("vertex weight range exceeds mesh weights".to_string());
285        }
286    }
287    let mesh = Md5Mesh {
288        shader,
289        vertices: std::mem::take(&mut vertices),
290        indices: triangles.into_iter().flatten().collect(),
291        weights,
292    };
293    let normals = compute_normals(&mesh, joints);
294    Ok(Md5Mesh {
295        vertices: normals,
296        ..mesh
297    })
298}

300/// Parse an MD5 mesh file (`parseMd5Mesh`).
301pub fn parse_md5_mesh(text: &str, source: &str) -> Result<Md5MeshFile, BinaryError> {
302    let mut tokens = ModelTokens::new(text, source);
303    let command_line = md5_header(&mut tokens)?;
304    tokens.expect("numJoints")?;
305    let joint_count = tokens.integer(1, 256)? as usize;
306    tokens.expect("numMeshes")?;
307    let mesh_count = tokens.integer(1, 32)? as usize;
308    tokens.expect("joints")?;
309    tokens.expect("{")?;
310    let mut joints = Vec::with_capacity(joint_count);
311    let mut bind_pose = Vec::with_capacity(joint_count);
312    for _ in 0..joint_count {
313        // Mesh parent order is metadata; model-space bind joints need no hierarchy traversal.
314        joints.push(Md5Joint {
315            name: tokens.token()?,
316            parent: tokens.integer(-1, joint_count as i32 - 1)?,
317            scale_positions: false,
318        });
319        bind_pose.push(SkeletonJointPose {
320            position: tokens.vector()?,
321            orientation: md5_quaternion(tokens.vector()?),
322            scale: 1.0,
323        });
324    }
325    tokens.expect("}")?;
326    let mut meshes = Vec::with_capacity(mesh_count);
327    for _ in 0..mesh_count {
328        meshes.push(read_mesh(&mut tokens, &bind_pose)?);
329    }
330    tokens.end()?;
331    Ok(Md5MeshFile {
332        source: source.to_string(),
333        command_line,
334        joints,
335        bind_pose,
336        meshes,
337    })
338}

340fn vector_normal(value: Vec3) -> Vec3 {
341    let (x, y, z) = (f64::from(value.x), f64::from(value.y), f64::from(value.z));
342    let length = (x * x + y * y + z * z).sqrt();
343    if length == 0.0 {
344        return value;
345    }
346    vec3((x / length) as f32, (y / length) as f32, (z / length) as f32)
347}

349fn compute_normals(mesh: &Md5Mesh, bind_pose: &[SkeletonJointPose]) -> Vec<Md5Vertex> {
350    let positions: Vec<Vec3> = mesh
351        .vertices
352        .iter()
353        .map(|vertex| {
354            let mut position = vec3(0.0, 0.0, 0.0);
355            for offset in 0..vertex.weights.count {
356                let weight = at(&mesh.weights, vertex.weights.first as usize + offset as usize, "weight");
357                let joint = at(bind_pose, weight.joint as usize, "joint");
358                let rotated = rotate_quaternion(joint.orientation, weight.position);
359                let world = add3(joint.position, rotated);
360                let bias = f64::from(weight.bias);
361                position = vec3(
362                    (f64::from(position.x) + f64::from(world.x) * bias) as f32,
363                    (f64::from(position.y) + f64::from(world.y) * bias) as f32,
364                    (f64::from(position.z) + f64::from(world.z) * bias) as f32,
365                );
366            }
367            position
368        })
369        .collect();
370    let key = |point: Vec3| format!("{}|{}|{}", point.x, point.y, point.z);
371    let mut normals: HashMap<String, Vec3> = HashMap::new();
372    for base in (0..mesh.indices.len()).step_by(3) {
373        let triangle = [mesh.indices[base], mesh.indices[base + 1], mesh.indices[base + 2]];
374        let a = at(&positions, triangle[0] as usize, "position");
375        let b = at(&positions, triangle[1] as usize, "position");
376        let c = at(&positions, triangle[2] as usize, "position");
377        let d1 = vector_normal(sub3(*c, *a));
378        let d2 = vector_normal(sub3(*b, *a));
379        let normal = vector_normal(vec3(
380            (f64::from(d1.y) * f64::from(d2.z) - f64::from(d1.z) * f64::from(d2.y)) as f32,
381            (f64::from(d1.z) * f64::from(d2.x) - f64::from(d1.x) * f64::from(d2.z)) as f32,
382            (f64::from(d1.x) * f64::from(d2.y) - f64::from(d1.y) * f64::from(d2.x)) as f32,
383        ));
384        let dot =
385            f64::from(d1.x) * f64::from(d2.x) + f64::from(d1.y) * f64::from(d2.y) + f64::from(d1.z) * f64::from(d2.z);
386        let angle = dot.clamp(-1.0, 1.0).acos();
387        let weighted = vec3(
388            (f64::from(normal.x) * angle) as f32,
389            (f64::from(normal.y) * angle) as f32,
390            (f64::from(normal.z) * angle) as f32,
391        );
392        for point in [a, b, c] {
393            let entry = normals.entry(key(*point)).or_insert(vec3(0.0, 0.0, 0.0));
394            *entry = add3(*entry, weighted);
395        }
396    }
397    for normal in normals.values_mut() {
398        *normal = vector_normal(*normal);
399    }
400    mesh.vertices
401        .iter()
402        .enumerate()
403        .map(|(index, vertex)| {
404            let world_normal = normals
405                .get(&key(positions[index]))
406                .copied()
407                .unwrap_or(vec3(0.0, 0.0, 0.0));
408            let mut normal = vec3(0.0, 0.0, 0.0);
409            for offset in 0..vertex.weights.count {
410                let weight = at(&mesh.weights, vertex.weights.first as usize + offset as usize, "weight");
411                let joint = at(bind_pose, weight.joint as usize, "joint");
412                let local = rotate_quaternion(conjugate_quaternion(joint.orientation), world_normal);
413                let bias = f64::from(weight.bias);
414                normal = vec3(
415                    (f64::from(normal.x) + bias * f64::from(local.x)) as f32,
416                    (f64::from(normal.y) + bias * f64::from(local.y)) as f32,
417                    (f64::from(normal.z) + bias * f64::from(local.z)) as f32,
418                );
419            }
420            Md5Vertex {
421                normal,
422                ..vertex.clone()
423            }
424        })
425        .collect()
426}

428#[derive(Debug, Clone, PartialEq)]
429enum JsonValue {
430    Null,
431    Bool(bool),
432    Number(f64),
433    String(String),
434    Array(Vec<JsonValue>),
435    Object(Vec<(String, JsonValue)>),
436}
437
438struct JsonParser {
439    chars: Vec<char>,
440    offset: usize,
441}
442
443impl JsonParser {
444    fn parse(text: &str) -> Option<JsonValue> {
445        let mut parser = Self {
446            chars: text.chars().collect(),
447            offset: 0,
448        };
449        let value = parser.value()?;
450        parser.whitespace();
451        if parser.offset != parser.chars.len() {
452            None
453        } else {
454            Some(value)
455        }
456    }
457
458    fn whitespace(&mut self) {
459        while self.offset < self.chars.len() && matches!(self.chars[self.offset], ' ' | '\t' | '\n' | '\r') {
460            self.offset += 1;
461        }
462    }
463
464    fn literal(&mut self, text: &str) -> bool {
465        let pattern: Vec<char> = text.chars().collect();
466        if self.chars[self.offset..].starts_with(&pattern) {
467            self.offset += pattern.len();
468            true
469        } else {
470            false
471        }
472    }
473
474    fn value(&mut self) -> Option<JsonValue> {
475        self.whitespace();
476        let character = *self.chars.get(self.offset)?;
477        match character {
478            'n' => self.literal("null").then_some(JsonValue::Null),
479            't' => self.literal("true").then_some(JsonValue::Bool(true)),
480            'f' => self.literal("false").then_some(JsonValue::Bool(false)),
481            '"' => self.string().map(JsonValue::String),
482            '[' => self.array(),
483            '{' => self.object(),
484            _ => self.number().map(JsonValue::Number),
485        }
486    }
487
488    fn string(&mut self) -> Option<String> {
489        if self.chars.get(self.offset) != Some(&'"') {
490            return None;
491        }
492        self.offset += 1;
493        let mut result = String::new();
494        while self.offset < self.chars.len() {
495            let character = self.chars[self.offset];
496            self.offset += 1;
497            match character {
498                '"' => return Some(result),
499                '\\' => {
500                    let escape = *self.chars.get(self.offset)?;
501                    self.offset += 1;
502                    match escape {
503                        '"' | '\\' | '/' => result.push(escape),
504                        'b' => result.push('\u{0008}'),
505                        'f' => result.push('\u{000c}'),
506                        'n' => result.push('\n'),
507                        'r' => result.push('\r'),
508                        't' => result.push('\t'),
509                        'u' => {
510                            let digits: String = self.chars.get(self.offset..self.offset + 4)?.iter().collect();
511                            let code = u32::from_str_radix(&digits, 16).ok()?;
512                            result.push(char::from_u32(code)?);
513                            self.offset += 4;
514                        }
515                        _ => return None,
516                    }
517                }
518                _ => {
519                    if (character as u32) < 0x20 {
520                        return None;
521                    }
522                    result.push(character);
523                }
524            }
525        }
526        None
527    }
528
529    fn number(&mut self) -> Option<f64> {
530        let start = self.offset;
531        if self.chars.get(self.offset) == Some(&'-') {
532            self.offset += 1;
533        }
534        let digits = |parser: &mut Self| {
535            let begin = parser.offset;
536            while parser.offset < parser.chars.len() && parser.chars[parser.offset].is_ascii_digit() {
537                parser.offset += 1;
538            }
539            parser.offset - begin
540        };
541        if self.chars.get(self.offset) == Some(&'0') {
542            self.offset += 1;
543        } else if digits(self) == 0 {
544            return None;
545        }
546        if self.chars.get(self.offset) == Some(&'.') {
547            self.offset += 1;
548            if digits(self) == 0 {
549                return None;
550            }
551        }
552        if matches!(self.chars.get(self.offset), Some('e' | 'E')) {
553            self.offset += 1;
554            if matches!(self.chars.get(self.offset), Some('+' | '-')) {
555                self.offset += 1;
556            }
557            if digits(self) == 0 {
558                return None;
559            }
560        }
561        if start == self.offset {
562            return None;
563        }
564        self.chars[start..self.offset].iter().collect::<String>().parse().ok()
565    }
566
567    fn array(&mut self) -> Option<JsonValue> {
568        self.offset += 1;
569        let mut items = Vec::new();
570        self.whitespace();
571        if self.chars.get(self.offset) == Some(&']') {
572            self.offset += 1;
573            return Some(JsonValue::Array(items));
574        }
575        loop {
576            items.push(self.value()?);
577            self.whitespace();
578            match self.chars.get(self.offset) {
579                Some(',') => {
580                    self.offset += 1;
581                }
582                Some(']') => {
583                    self.offset += 1;
584                    return Some(JsonValue::Array(items));
585                }
586                _ => return None,
587            }
588        }
589    }
590
591    fn object(&mut self) -> Option<JsonValue> {
592        self.offset += 1;
593        let mut entries = Vec::new();
594        self.whitespace();
595        if self.chars.get(self.offset) == Some(&'}') {
596            self.offset += 1;
597            return Some(JsonValue::Object(entries));
598        }
599        loop {
600            self.whitespace();
601            let key = self.string()?;
602            self.whitespace();
603            if self.chars.get(self.offset) != Some(&':') {
604                return None;
605            }
606            self.offset += 1;
607            entries.push((key, self.value()?));
608            self.whitespace();
609            match self.chars.get(self.offset) {
610                Some(',') => {
611                    self.offset += 1;
612                }
613                Some('}') => {
614                    self.offset += 1;
615                    return Some(JsonValue::Object(entries));
616                }
617                _ => return None,
618            }
619        }
620    }
621}

623fn js_number(text: &str) -> f64 {
624    let trimmed =
625        text.trim_matches(|character: char| matches!(character, '\t' | '\n' | '\u{000b}' | '\u{000c}' | '\r' | ' '));
626    if trimmed.is_empty() {
627        return 0.0;
628    }
629    trimmed.parse().unwrap_or(f64::NAN)
630}

634/// Optional scale failures are diagnostics, matching q2repro's fallback to
635/// unit scales (`parseScales`).
636fn parse_scales(source: Option<&Md5ScaleSource>, hierarchy: &[Md5HierarchyJoint], frame_count: usize) -> ParsedScales {
637    let mut scales: HashMap<(usize, usize), f32> = HashMap::new();
638    let mut scale_positions: HashSet<usize> = HashSet::new();
639    let mut diagnostics = Vec::new();
640    if let Some(source) = source {
641        let root = JsonParser::parse(&source.text);
642        let entries = match root {
643            Some(JsonValue::Object(entries)) => Some(entries),
644            _ => None,
645        };
646        match entries {
647            None => diagnostics.push(format!("{}: Invalid JSON scale object", source.source)),
648            Some(entries) => {
649                for (name, entry) in &entries {
650                    let JsonValue::Object(values) = entry else {
651                        diagnostics.push(format!("{}: Invalid scale entry for {name}", source.source));
652                        break;
653                    };
654                    let joint = hierarchy.iter().position(|joint| joint.name == *name);
655                    let Some(joint) = joint else {
656                        diagnostics.push(format!("{}: No such joint {name}", source.source));
657                        continue;
658                    };
659                    for (key, value) in values {
660                        if key == "scale_positions" {
661                            if *value == JsonValue::Bool(true) {
662                                scale_positions.insert(joint);
663                            }
664                        } else {
665                            let frame = js_number(key);
666                            let valid = !key.is_empty()
667                                && frame.fract() == 0.0
668                                && frame >= 0.0
669                                && frame < frame_count as f64
670                                && matches!(value, JsonValue::Number(_));
671                            let rounded = match value {
672                                JsonValue::Number(number) => *number as f32,
673                                _ => f32::NAN,
674                            };
675                            if valid && rounded.is_finite() {
676                                scales.insert((frame as usize, joint), rounded);
677                            } else {
678                                diagnostics.push(format!("{}: Invalid frame scale {name}/{key}", source.source));
679                            }
680                        }
681                    }
682                }
683            }
684        }
685    }
686    let hierarchy = hierarchy
687        .iter()
688        .enumerate()
689        .map(|(index, joint)| Md5HierarchyJoint {
690            name: joint.name.clone(),
691            parent: joint.parent,
692            flags: joint.flags,
693            start_index: joint.start_index,
694            scale_positions: scale_positions.contains(&index),
695        })
696        .collect();
697    (hierarchy, scales, diagnostics)
698}

700/// Parse an MD5 animation file (`parseMd5Anim`).
701pub fn parse_md5_anim(
702    text: &str,
703    source: &str,
704    scale_source: Option<&Md5ScaleSource>,
705) -> Result<Md5Animation, BinaryError> {
706    let mut tokens = ModelTokens::new(text, source);
707    let command_line = md5_header(&mut tokens)?;
708    tokens.expect("numFrames")?;
709    let frame_count = tokens.integer(1, 65535)? as usize;
710    tokens.expect("numJoints")?;
711    let joint_count = tokens.integer(1, 256)? as usize;
712    tokens.expect("frameRate")?;
713    let frame_rate = tokens.integer(1, 1000)?;
714    tokens.expect("numAnimatedComponents")?;
715    let component_count = tokens.integer(0, joint_count as i32 * 6)? as usize;
716    tokens.expect("hierarchy")?;
717    tokens.expect("{")?;
718    let mut raw_hierarchy = Vec::with_capacity(joint_count);
719    for index in 0..joint_count {
720        let name = tokens.token()?;
721        let parent = tokens.integer(-1, index as i32 - 1)?;
722        let flags = tokens.integer(0, 63)? as u32;
723        let start_index = tokens.integer(0, component_count as i32)? as usize;
724        let mut animated = 0;
725        for bit in 0..6 {
726            if flags & (1 << bit) != 0 {
727                animated += 1;
728            }
729        }
730        if start_index + animated > component_count {
731            return tokens.fail("animated joint components exceed frame".to_string());
732        }
733        raw_hierarchy.push(Md5HierarchyJoint {
734            name,
735            parent,
736            flags,
737            start_index,
738            scale_positions: false,
739        });
740    }
741    tokens.expect("}")?;
742    tokens.expect("bounds")?;
743    tokens.expect("{")?;
744    let mut bounds = Vec::with_capacity(frame_count);
745    for _ in 0..frame_count {
746        bounds.push(Bounds {
747            min: tokens.vector()?,
748            max: tokens.vector()?,
749        });
750    }
751    tokens.expect("}")?;
752    tokens.expect("baseframe")?;
753    tokens.expect("{")?;
754    let mut base_frame = Vec::with_capacity(joint_count);
755    for _ in 0..joint_count {
756        base_frame.push(SkeletonJointPose {
757            position: tokens.vector()?,
758            orientation: md5_quaternion(tokens.vector()?),
759            scale: 1.0,
760        });
761    }
762    tokens.expect("}")?;
763    let (hierarchy, scales, diagnostics) = parse_scales(scale_source, &raw_hierarchy, frame_count);
764    let raw_frames = indexed_records(&mut tokens, frame_count, "frame", |tokens| {
765        tokens.expect("{")?;
766        let mut frame = Vec::with_capacity(component_count);
767        for _ in 0..component_count {
768            frame.push(tokens.float()?);
769        }
770        tokens.expect("}")?;
771        Ok(frame)
772    })?;
773    tokens.end()?;
774    let mut frames = Vec::with_capacity(frame_count);
775    for (frame, components) in raw_frames.iter().enumerate() {
776        let mut local_joints = Vec::with_capacity(joint_count);
777        let mut joints = Vec::with_capacity(joint_count);
778        for index in 0..joint_count {
779            let info = at(&hierarchy, index, "hierarchy joint");
780            let base = at(&base_frame, index, "base joint");
781            let mut component = info.start_index;
782            let mut animated = |bit: u32, fallback: f32| {
783                if info.flags & bit == 0 {
784                    fallback
785                } else {
786                    let value = components[component];
787                    component += 1;
788                    value
789                }
790            };
791            let position = vec3(
792                animated(1, base.position.x),
793                animated(2, base.position.y),
794                animated(4, base.position.z),
795            );
796            let orientation = md5_quaternion(vec3(
797                animated(8, base.orientation.x),
798                animated(16, base.orientation.y),
799                animated(32, base.orientation.z),
800            ));
801            let scale = scales.get(&(frame, index)).copied().unwrap_or(1.0);
802            local_joints.push(SkeletonJointPose {
803                position,
804                orientation,
805                scale,
806            });
807            let scaled_position = if info.scale_positions {
808                scale_position(position, scale)
809            } else {
810                position
811            };
812            if info.parent < 0 {
813                joints.push(SkeletonJointPose {
814                    position: scaled_position,
815                    orientation,
816                    scale,
817                });
818            } else {
819                let parent = at(&joints, info.parent as usize, "parent joint");
820                joints.push(SkeletonJointPose {
821                    position: add3(parent.position, rotate_quaternion(parent.orientation, scaled_position)),
822                    orientation: normalize_quaternion(multiply_quaternion(parent.orientation, orientation)),
823                    scale,
824                });
825            }
826        }
827        frames.push(Md5AnimationFrame {
828            bounds: bounds[frame],
829            components: components.clone(),
830            local_joints,
831            joints,
832        });
833    }
834    Ok(Md5Animation {
835        source: source.to_string(),
836        command_line,
837        frame_rate,
838        hierarchy,
839        base_frame,
840        frames,
841        scale_source: scale_source.map(|source| source.source.clone()),
842        diagnostics,
843    })
844}

847    vec3(position.x * scale, position.y * scale, position.z * scale)

850/// Join a mesh file with its animation (`createMd5Model`).
851///
852/// # Panics
853///
854/// Panics when the joint counts differ.
855#[must_use]
856pub fn create_md5_model(mesh: Md5MeshFile, animation: Md5Animation, skin_selection: SkinSelection) -> DecodedMd5Model {
857    if mesh.joints.len() != animation.hierarchy.len() {
858        panic!("{}: mesh and animation joint counts differ", mesh.source);
859    }
860    DecodedMd5Model {
861        mesh_source: mesh.source.clone(),
862        mesh_command_line: mesh.command_line.clone(),
863        mesh_joints: mesh.joints.clone(),
864        bind_pose: mesh.bind_pose.clone(),
865        frame_rate: animation.frame_rate,
866        joints: animation.hierarchy.clone(),
867        frames: animation.frames.clone(),
868        meshes: mesh.meshes.clone(),
869        animation,
870        skin_selection,
871    }
872}

874/// Interpolate joint poses (`sampleMd5Pose`).
875///
876/// Positions and orientations blend; Q2 uses the new frame's scale
877/// directly.
878///
879/// # Panics
880///
881/// Panics on invalid frame selections.
882#[must_use]
883pub fn sample_md5_pose(
884    frames: &[Md5AnimationFrame],
885    frame: i32,
886    previous_frame: i32,
887    back_lerp: f32,
888) -> Vec<SkeletonJointPose> {
889    if frame < 0 || previous_frame < 0 || !back_lerp.is_finite() {
890        panic!("Invalid MD5 frame selection");
891    }
892    let current = &at(frames, frame as usize % frames.len(), "MD5 frame").joints;
893    if back_lerp == 0.0 || frame == previous_frame {
894        return current.to_vec();
895    }
896    let previous = &at(frames, previous_frame as usize % frames.len(), "MD5 previous frame").joints;
897    let back = f64::from(back_lerp);
898    let front = 1.0 - back;
899    current
900        .iter()
901        .enumerate()
902        .map(|(index, joint)| {
903            let old = at(previous, index, "previous joint");
904            SkeletonJointPose {
905                position: vec3(
906                    (f64::from(old.position.x) * back + f64::from(joint.position.x) * front) as f32,
907                    (f64::from(old.position.y) * back + f64::from(joint.position.y) * front) as f32,
908                    (f64::from(old.position.z) * back + f64::from(joint.position.z) * front) as f32,
909                ),
910                orientation: slerp_quaternion(old.orientation, joint.orientation, back_lerp),
911                scale: joint.scale,
912            }
913        })
914        .collect()
915}

917/// Skin one MD5 mesh (`skinMd5Mesh`).
918#[must_use]
919pub fn skin_md5_mesh(mesh: &Md5Mesh, joints: &[SkeletonJointPose]) -> Vec<Md5SkinnedVertex> {
920    let mut rotations: Vec<Option<crate::quaternion::QuaternionRotationRows>> = Vec::new();
921    mesh.vertices
922        .iter()
923        .map(|vertex| {
924            let mut position = vec3(0.0, 0.0, 0.0);
925            let mut normal = vec3(0.0, 0.0, 0.0);
926            for offset in 0..vertex.weights.count {
927                let weight = at(&mesh.weights, vertex.weights.first as usize + offset as usize, "weight");
928                let joint = at(joints, weight.joint as usize, "joint");
929                while rotations.len() <= weight.joint as usize {
930                    rotations.push(None);
931                }
932                let rows = *rotations[weight.joint as usize]
933                    .get_or_insert_with(|| quaternion_rotation_rows(joint.orientation));
934                let rotated = rotate_quaternion_rows(&rows, weight.position);
935                let scale = f64::from(joint.scale);
936                let point = vec3(
937                    (f64::from(joint.position.x) + scale * f64::from(rotated.x)) as f32,
938                    (f64::from(joint.position.y) + scale * f64::from(rotated.y)) as f32,
939                    (f64::from(joint.position.z) + scale * f64::from(rotated.z)) as f32,
940                );
941                let direction = rotate_quaternion_rows(&rows, vertex.normal);
942                let bias = f64::from(weight.bias);
943                position = vec3(
944                    (f64::from(position.x) + bias * f64::from(point.x)) as f32,
945                    (f64::from(position.y) + bias * f64::from(point.y)) as f32,
946                    (f64::from(position.z) + bias * f64::from(point.z)) as f32,
947                );
948                normal = vec3(
949                    (f64::from(normal.x) + bias * f64::from(direction.x)) as f32,
950                    (f64::from(normal.y) + bias * f64::from(direction.y)) as f32,
951                    (f64::from(normal.z) + bias * f64::from(direction.z)) as f32,
952                );
953            }
954            Md5SkinnedVertex { position, normal }
955        })
956        .collect()
957}
