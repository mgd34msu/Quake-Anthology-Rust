1//! Quake II alias model (MD2 v8) parser with GL commands and animation.
2//!
3//! Donor provenance: `parseMd2` and `decodeMd2Commands` in
4//! `src/formats/q12-model/md2.ts`, with shared readers from
5//! `src/formats/q12-model/common.ts` (see [`crate::common`]), normals
6//! from [`crate::normals`], and pose sampling, interpolation, and
7//! geometry builders from `src/formats/q12-model/animation.ts`.
8//!
9//! Skins and decoded vertices are owned; sections are read sequentially
10//! through sub-readers, borrowing nothing past validation.

100fn read_commands(
101    reader: &mut BinaryReader<'_>,
102    source: &str,
103    vertex_count: i32,
104) -> Result<Vec<Md2Command>, BinaryError> {
105    let mut result = Vec::new();
106    if reader.length() == 0 {
107        return Ok(result);
108    }
109    while reader.remaining() > 0 {
110        let command = reader.i32()?;
111        if command == 0 {
112            return Ok(result);
113        }
114        let count = (command as i64).abs();
115        if count < 3 {
116            return fail(
117                reader,
118                source,
119                "GL strip/fan must have at least three vertices".to_string(),
120            );
121        }
122        reader.section(reader.offset(), count as usize * 12)?;
123        let mut vertices = Vec::with_capacity(count as usize);
124        for _ in 0..count {
125            let tex_coord = [reader.finite_f32()?, reader.finite_f32()?];
126            let vertex = reader.i32()?;
127            vertices.push(Md2CommandVertex {
128                tex_coord,
129                vertex: check_index(reader, source, vertex, vertex_count, "GL vertex")?,
130            });
131        }
132        result.push(Md2Command {
133            strip: command > 0,
134            vertices,
135        });
136    }
137    fail(reader, source, "GL command list lacks terminator".to_string())
138}

140/// Decode the model's GL command words (`decodeMd2Commands`).
141///
142/// Float UV bit patterns remain intact in the round trip through bytes.
143pub fn decode_md2_commands(model: &Md2Model) -> Result<Vec<Md2Command>, BinaryError> {
144    let source = "<md2 GL commands>";
145    let Some(frame) = model.frames.first() else {
146        return Err(BinaryError {
147            input: source.to_string(),
148            offset: 0,
149            message: "MD2 has no frames".to_string(),
150        });
151    };
152    let mut data = Vec::with_capacity(model.gl_commands.len() * 4);
153    for word in &model.gl_commands {
154        data.extend_from_slice(&word.to_le_bytes());
155    }
156    read_commands(
157        &mut BinaryReader::new(&data, source),
158        source,
159        frame.vertices.len() as i32,
160    )
161}

163fn section<'a>(
164    reader: &mut BinaryReader<'a>,
165    source: &str,
166    offset: i32,
167    count: i32,
168    stride: i64,
169    end: i32,
170) -> Result<BinaryReader<'a>, BinaryError> {
171    let size = count as i64 * stride;
172    if (offset as i64) < MD2_HEADER_SIZE || size > i64::from(end) || (offset as i64) > i64::from(end) - size {
173        return fail(
174            reader,
175            source,
176            format!("MD2 section {offset}+{size} exceeds header/model bounds"),
177        );
178    }
179    reader.section(offset as usize, size as usize)
180}

182/// Parse an MD2 v8 model (`parseMd2`).
183pub fn parse_md2(data: &[u8], source: &str) -> Result<Md2Model, BinaryError> {
184    let mut reader = BinaryReader::new(data, source);
185    reader.expect_magic("IDP2")?;
186    version(&mut reader, source, 8)?;
187    let skin_width = count(&mut reader, source, "skin width", 1)?;
188    let skin_height = count(&mut reader, source, "skin height", 1)?;
189    let frame_size = count(&mut reader, source, "frame size", 1)?;
190    let skin_count = count(&mut reader, source, "skins", 0)?;
191    let vertex_count = count(&mut reader, source, "vertices", 1)?;
192    let coordinate_count = count(&mut reader, source, "texture coordinates", 1)?;
193    let triangle_count = count(&mut reader, source, "triangles", 1)?;
194    let command_count = count(&mut reader, source, "GL command words", 0)?;
195    let frame_count = count(&mut reader, source, "frames", 1)?;
196    let skin_offset = reader.i32()?;
197    let coordinate_offset = reader.i32()?;
198    let triangle_offset = reader.i32()?;
199    let frame_offset = reader.i32()?;
200    let command_offset = reader.i32()?;
201    let end = reader.i32()?;
202    if end < MD2_HEADER_SIZE as i32 || end as usize > reader.length() {
203        return fail(&reader, source, format!("invalid model end {end}"));
204    }
205    if (frame_size as i64) < 40 + vertex_count as i64 * 4 {
206        return fail(&reader, source, "frame size is smaller than its vertices".to_string());
207    }
208    let mut skin_reader = section(&mut reader, source, skin_offset, skin_count, 64, end)?;
209    let mut coordinate_reader = section(&mut reader, source, coordinate_offset, coordinate_count, 4, end)?;
210    let mut triangle_reader = section(&mut reader, source, triangle_offset, triangle_count, 12, end)?;
211    let mut frame_reader = section(&mut reader, source, frame_offset, frame_count, frame_size as i64, end)?;
212    let mut command_reader = section(&mut reader, source, command_offset, command_count, 4, end)?;
213    let mut skins = Vec::with_capacity(skin_count as usize);
214    for _ in 0..skin_count {
215        skins.push(skin_reader.fixed_byte_string(64)?);
216    }
217    let mut texture_coordinates = Vec::with_capacity(coordinate_count as usize);
218    for _ in 0..coordinate_count {
219        texture_coordinates.push(Md2TexCoord {
220            x: i32::from(coordinate_reader.i16()?),
221            y: i32::from(coordinate_reader.i16()?),
222        });
223    }
224    let mut triangles = Vec::with_capacity(triangle_count as usize);
225    for _ in 0..triangle_count {
226        let mut vertices = [0u32; 3];
227        let mut tex_coords = [0u32; 3];
228        for slot in &mut vertices {
229            let value = triangle_reader.u16()?;
230            *slot = check_index(&triangle_reader, source, i32::from(value), vertex_count, "vertex")?;
231        }
232        for slot in &mut tex_coords {
233            let value = triangle_reader.u16()?;
234            *slot = check_index(
235                &triangle_reader,
236                source,
237                i32::from(value),
238                coordinate_count,
239                "texture coordinate",
240            )?;
241        }
242        triangles.push(Md2Triangle { vertices, tex_coords });
243    }
244    let mut frames = Vec::with_capacity(frame_count as usize);
245    for frame in 0..frame_count {
246        frame_reader.seek(frame as usize * frame_size as usize)?;
247        let scale = vector(&mut frame_reader)?;
248        let translation = vector(&mut frame_reader)?;
249        let name = frame_reader.fixed_byte_string(16)?;
250        let mut vertices = Vec::with_capacity(vertex_count as usize);
251        let mut compressed_vertices = Vec::with_capacity(vertex_count as usize);
252        for _ in 0..vertex_count {
253            let packed = packed_vertex(&mut frame_reader)?;
254            check_index(&frame_reader, source, i32::from(packed.normal_index), 162, "normal")?;
255            compressed_vertices.push(packed);
256            vertices.push(ModelVertex {
257                position: packed_position(packed, scale, translation),
258                normal: alias_normal(packed.normal_index).unwrap_or([0.0, 0.0, 1.0]),
259            });
260        }
261        let positions: Vec<[f32; 3]> = vertices.iter().map(|vertex| vertex.position).collect();
262        frames.push(Md2Frame {
263            name,
264            scale,
265            translation,
266            origin: [0.0, 0.0, 0.0],
267            vertices,
268            compressed_vertices,
269            bounds: bounds_from_points(&positions),
270        });
271    }
272    read_commands(&mut command_reader, source, vertex_count)?;
273    command_reader.seek(0)?;
274    let mut gl_commands = Vec::with_capacity(command_count as usize);
275    for _ in 0..command_count {
276        gl_commands.push(command_reader.i32()?);
277    }
278    let bounds = union_bounds(&frames.iter().map(|frame| frame.bounds).collect::<Vec<_>>());
279    Ok(Md2Model {
280        skin_width,
281        skin_height,
282        skins,
283        texture_coordinates,
284        triangles,
285        frames,
286        gl_commands,
287        bounds,
288    })
289}

295/// Select the frame active at a time (`sampleTimedFrame`).
296///
297/// Q1 stores cumulative endpoints, not durations: a boundary selects the
298/// next frame.
299///
300/// # Panics
301///
302/// Panics when either time argument is non-finite.
303#[must_use]
304pub fn sample_timed_frame<T>(frames: &TimedFrames<T>, time_seconds: f64, sync_base: f64) -> &T {
305    if !time_seconds.is_finite() || !sync_base.is_finite() {
306        panic!("Animation time must be finite");
307    }
308    match frames {
309        TimedFrames::Single(frame) => frame,
310        TimedFrames::Group(frames) => {
311            let last = at(frames, frames.len() - 1, "group frame");
312            let time = time_seconds + sync_base;
313            let endpoint = f64::from(last.interval_seconds);
314            let target = time - (time / endpoint).trunc() * endpoint;
315            for item in frames {
316                if f64::from(item.interval_seconds) > target {
317                    return &item.frame;
318                }
319            }
320            &last.frame
321        }
322    }
323}

325/// Blend two alias poses (`interpolateAliasFrames`).
326///
327/// Q1 and Q2 light the current pose's normal while interpolating vertex
328/// positions.
329///
330/// # Panics
331///
332/// Panics when `back_lerp` is outside `0..=1` or the poses differ in
333/// length.
334#[must_use]
335pub fn interpolate_alias_frames(
336    current: &[ModelVertex],
337    previous: &[ModelVertex],
338    back_lerp: f32,
339    previous_origin_delta: [f32; 3],
340) -> Vec<ModelVertex> {
341    if !back_lerp.is_finite() || back_lerp < 0.0 || back_lerp > 1.0 {
342        panic!("backLerp must be in 0..1");
343    }
344    if current.len() != previous.len() {
345        panic!("Alias poses have different vertex counts");
346    }
347    let back = f64::from(back_lerp);
348    let front = 1.0 - back;
349    current
350        .iter()
351        .zip(previous.iter())
352        .map(|(vertex, old)| {
353            let position = [
354                (f64::from(vertex.position[0]) * front
355                    + (f64::from(old.position[0]) + f64::from(previous_origin_delta[0])) * back) as f32,
356                (f64::from(vertex.position[1]) * front
357                    + (f64::from(old.position[1]) + f64::from(previous_origin_delta[1])) * back) as f32,
358                (f64::from(vertex.position[2]) * front
359                    + (f64::from(old.position[2]) + f64::from(previous_origin_delta[2])) * back) as f32,
360            ];
361            ModelVertex {
362                position,
363                normal: vertex.normal,
364            }
365        })
366        .collect()
367}

389fn md2_tex_coord(model: &Md2Model, coordinate: Md2TexCoord) -> [f32; 2] {
390    [
391        ((f64::from(coordinate.x) + 0.5) / f64::from(model.skin_width)) as f32,
392        ((f64::from(coordinate.y) + 0.5) / f64::from(model.skin_height)) as f32,
393    ]
394}

396/// Expand MD2 triangles into corners (`buildMd2Geometry`).
397///
398/// # Panics
399///
400/// Panics when the pose is shorter than the model's vertex count.
401#[must_use]
402pub fn build_md2_geometry(model: &Md2Model, pose: &[ModelVertex]) -> Md2Geometry {
403    let mut vertices = Vec::with_capacity(model.triangles.len() * 3);
404    let mut indices = Vec::with_capacity(model.triangles.len() * 3);
405    for triangle in &model.triangles {
406        for corner in 0..3 {
407            let index = triangle.vertices[corner] as usize;
408            let vertex = at(pose, index, "pose vertex");
409            let coordinate = triangle.tex_coords[corner] as usize;
410            let tex_coord = md2_tex_coord(model, *at(&model.texture_coordinates, coordinate, "texture coordinate"));
411            indices.push(vertices.len() as u32);
412            vertices.push(Md2MeshVertex {
413                position: vertex.position,
414                normal: vertex.normal,
415                tex_coord,
416            });
417        }
418    }
419    Md2Geometry { vertices, indices }
420}

431/// Map MD2 corners into a consumer's geometry (`mapMd2Geometry`).
432///
433/// # Panics
434///
435/// Panics when the pose is shorter than the model's vertex count.
436pub fn map_md2_geometry<T>(
437    model: &Md2Model,
438    pose: &[ModelVertex],
439    map: impl Fn(&ModelVertex, [f32; 2], usize) -> T,
440) -> Md2MappedGeometry<T> {
441    let mut vertices = Vec::with_capacity(model.triangles.len() * 3);
442    let mut indices = Vec::with_capacity(model.triangles.len() * 3);
443    for triangle in &model.triangles {
444        for corner in 0..3 {
445            let vertex = at(pose, triangle.vertices[corner] as usize, "pose vertex");
446            let coordinate = triangle.tex_coords[corner] as usize;
447            let tex_coord = md2_tex_coord(model, *at(&model.texture_coordinates, coordinate, "texture coordinate"));
448            indices.push(vertices.len() as u32);
449            let corner_index = vertices.len();
450            vertices.push(map(vertex, tex_coord, corner_index));
451        }
452    }
453    Md2MappedGeometry { vertices, indices }
454}
