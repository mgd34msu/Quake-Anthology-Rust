33/// BSP codec family (`classifyBsp` in `src/formats/bsp-kind.ts`).
34///
35/// Container ownership does not determine a map's stored codec.
36#[derive(Debug, Clone, Copy, PartialEq, Eq)]
37pub enum BspKind {
38    /// Quake BSP29 / BSP2 / 2PSB / Quake64.
39    Q1,
40    /// Quake II IBSP/QBSP version 38.
41    Q2,
42    /// Quake III IBSP versions 44/46.
43    Q3,
44}
45
46/// Classify a map by its stored codec (`classifyBsp`).
47pub fn classify_bsp(data: &[u8], source: &str) -> Result<BspKind, BinaryError> {
48    let mut reader = BinaryReader::new(data, source);
49    let magic = reader.u32()?;
50    if magic == 29 || magic == 0x3250_5342 || magic == 0x4253_5032 || magic == 0x5136_3420 {
51        return Ok(BspKind::Q1);
52    }
53    if magic == 0x5053_4249 || magic == 0x5053_4251 {
54        let version = reader.u32()?;
55        if version == 38 {
56            return Ok(BspKind::Q2);
57        }
58        if magic == 0x5053_4249 && (version == 44 || version == 46) {
59            return Ok(BspKind::Q3);
60        }
61        return Err(BinaryError {
62            input: source.to_string(),
63            offset: 4,
64            message: format!("unsupported BSP version {version}"),
65        });
66    }
67    Err(BinaryError {
68        input: source.to_string(),
69        offset: 0,
70        message: format!("unsupported BSP identifier {magic}"),
71    })
72}

74#[cfg(test)]
75mod tests {
76    use super::*;
77
78    #[test]
79    fn classifies_families() {
80        let mut q1 = [0u8; 8];
81        q1[0] = 29;
82        assert_eq!(classify_bsp(&q1, "<test>").unwrap(), BspKind::Q1);
83        let quake64 = [0x20, 0x34, 0x36, 0x51, 0, 0, 0, 0];
84        assert_eq!(classify_bsp(&quake64, "<test>").unwrap(), BspKind::Q1);
85        let q2 = [b'I', b'B', b'S', b'P', 38, 0, 0, 0];
86        assert_eq!(classify_bsp(&q2, "<test>").unwrap(), BspKind::Q2);
87        let qbsp = [b'Q', b'B', b'S', b'P', 38, 0, 0, 0];
88        assert_eq!(classify_bsp(&qbsp, "<test>").unwrap(), BspKind::Q2);
89        let q3 = [b'I', b'B', b'S', b'P', 46, 0, 0, 0];
90        assert_eq!(classify_bsp(&q3, "<test>").unwrap(), BspKind::Q3);
91        let q3_test = [b'I', b'B', b'S', b'P', 44, 0, 0, 0];
92        assert_eq!(classify_bsp(&q3_test, "<test>").unwrap(), BspKind::Q3);
93        let bad_version = [b'I', b'B', b'S', b'P', 45, 0, 0, 0];
94        let error = classify_bsp(&bad_version, "<test>").unwrap_err();
95        assert_eq!(error.message, "unsupported BSP version 45");
96        assert_eq!(error.offset, 4);
97        let error = classify_bsp(b"XXXX....", "<test>").unwrap_err();
98        assert!(
99            error.message.contains("unsupported BSP identifier"),
100            "{}",
101            error.message
102        );
103        assert_eq!(error.offset, 0);
104    }
105}
