1//! Quake III skeletal model (MD4 v1) parser and skinning.
2//!
3//! Donor provenance: `src/formats/q3-model/md4.ts` (loading from
4//! `qfiles.h` and `tr_model.c`, bone indexing from `tr_animation.c`,
5//! skinning from `RB_SurfaceAnim`). The dot-product order matches
6//! [`qa_core::math::dot3`], which this port reuses.

132fn md4_fail(source: &str, offset: usize, message: String) -> BinaryError {
133    BinaryError {
134        input: source.to_string(),
135        offset,
136        message,
137    }
138}

140fn md4_range(source: &str, offset: i64, length: i64, start: i64, end: i64, field: usize) -> Result<(), BinaryError> {
141    if offset < start || length < 0 || offset > end - length {
142        return Err(md4_fail(
143            source,
144            field,
145            format!("MD4 range {offset}+{length} exceeds {start}..{end}"),
146        ));
147    }
148    Ok(())
149}

151fn md4_count(reader: &mut BinaryReader<'_>, source: &str, maximum: i32) -> Result<i32, BinaryError> {
152    let offset = reader.offset();
153    let value = reader.i32()?;
154    if value < 0 || value > maximum {
155        return Err(md4_fail(
156            source,
157            offset,
158            format!("MD4 count {value} outside 0..{maximum}"),
159        ));
160    }
161    Ok(value)
162}
163
164fn md4_index(reader: &mut BinaryReader<'_>, source: &str, maximum: i32, label: &str) -> Result<u32, BinaryError> {
165    let offset = reader.offset();
166    let value = reader.i32()?;
167    if value < 0 || value >= maximum {
168        return Err(md4_fail(
169            source,
170            offset,
171            format!("{label} {value} outside 0..{}", maximum - 1),
172        ));
173    }
174    Ok(value as u32)
175}

177fn md4_string(reader: &mut BinaryReader<'_>) -> Result<String, BinaryError> {
178    let bytes = reader.bytes(64)?;
179    let mut result = String::new();
180    for byte in bytes {
181        if byte == 0 {
182            break;
183        }
184        result.push(char::from(byte));
185    }
186    Ok(result)
187}

189fn md4_vec3(reader: &mut BinaryReader<'_>) -> Result<Vec3, BinaryError> {
190    Ok(vec3(reader.finite_f32()?, reader.finite_f32()?, reader.finite_f32()?))
191}

193fn md4_row(reader: &mut BinaryReader<'_>) -> Result<Vec4, BinaryError> {
194    Ok(vec4(
195        reader.finite_f32()?,
196        reader.finite_f32()?,
197        reader.finite_f32()?,
198        reader.finite_f32()?,
199    ))
200}

202fn parse_surface(
203    reader: &mut BinaryReader<'_>,
204    source: &str,
205    surface_offset: i64,
206    lod_end: i64,
207    num_bones: i32,
208) -> Result<(Md4Surface, i64), BinaryError> {
209    md4_range(
210        source,
211        surface_offset,
212        SURFACE_SIZE,
213        HEADER_SIZE,
214        lod_end,
215        surface_offset as usize,
216    )?;
217    reader.seek(surface_offset as usize)?;
218    reader.i32()?; // R_LoadMD4 replaces the disk ident with SF_MD4 without checking it.
219    let name = md4_string(reader)?.to_ascii_lowercase();
220    let shader = md4_string(reader)?;
221    reader.i32()?; // Replaced by actual shader registration, outside the binary reader.
222    let ofs_header = reader.i32()?;
223    if surface_offset + i64::from(ofs_header) != 0 {
224        return Err(md4_fail(
225            source,
226            (surface_offset + 136) as usize,
227            "MD4 surface does not point back to its model header".to_string(),
228        ));
229    }
230    let num_verts = md4_count(reader, source, 1000)?;
231    let ofs_verts = reader.i32()?;
232    let num_triangles = md4_count(reader, source, 2000)?;
233    let ofs_triangles = reader.i32()?;
234    let num_bone_references = md4_count(reader, source, i32::MAX)?;
235    let ofs_bone_references = reader.i32()?;
236    let ofs_end = reader.i32()?;
237    md4_range(
238        source,
239        surface_offset,
240        i64::from(ofs_end),
241        surface_offset,
242        lod_end,
243        (surface_offset + 164) as usize,
244    )?;
245    if ofs_end < SURFACE_SIZE as i32 {
246        return Err(md4_fail(
247            source,
248            (surface_offset + 164) as usize,
249            "MD4 surface end precedes its header".to_string(),
250        ));
251    }
252    let end = surface_offset + i64::from(ofs_end);
253    if num_verts > 0 {
254        md4_range(
255            source,
256            i64::from(ofs_verts),
257            i64::from(num_verts) * 24,
258            SURFACE_SIZE,
259            i64::from(ofs_end),
260            (surface_offset + 144) as usize,
261        )?;
262    }
263    if num_triangles > 0 {
264        md4_range(
265            source,
266            i64::from(ofs_triangles),
267            i64::from(num_triangles) * 12,
268            SURFACE_SIZE,
269            i64::from(ofs_end),
270            (surface_offset + 152) as usize,
271        )?;
272    }
273    if num_bone_references > 0 {
274        md4_range(
275            source,
276            i64::from(ofs_bone_references),
277            i64::from(num_bone_references) * 4,
278            SURFACE_SIZE,
279            i64::from(ofs_end),
280            (surface_offset + 160) as usize,
281        )?;
282    }
283
284    let mut triangles = Vec::with_capacity(num_triangles as usize);
285    if num_triangles > 0 {
286        reader.seek((surface_offset + i64::from(ofs_triangles)) as usize)?;
287    }
288    for _ in 0..num_triangles {
289        triangles.push(Md4Triangle {
290            indices: [
291                md4_index(reader, source, num_verts, "triangle vertex")?,
292                md4_index(reader, source, num_verts, "triangle vertex")?,
293                md4_index(reader, source, num_verts, "triangle vertex")?,
294            ],
295        });
296    }
297    let mut bone_references = Vec::with_capacity(num_bone_references as usize);
298    if num_bone_references > 0 {
299        reader.seek((surface_offset + i64::from(ofs_bone_references)) as usize)?;
300    }
301    for _ in 0..num_bone_references {
302        bone_references.push(md4_index(reader, source, num_bones, "bone reference")?);
303    }
304
305    let mut vertices = Vec::with_capacity(num_verts as usize);
306    if num_verts > 0 {
307        reader.seek((surface_offset + i64::from(ofs_verts)) as usize)?;
308    }
309    for _ in 0..num_verts {
310        md4_range(
311            source,
312            reader.offset() as i64,
313            24,
314            surface_offset + SURFACE_SIZE,
315            end,
316            reader.offset(),
317        )?;
318        let normal = md4_vec3(reader)?;
319        let tex_coords = vec2(reader.finite_f32()?, reader.finite_f32()?);
320        let num_weights = md4_count(reader, source, i32::MAX)?;
321        md4_range(
322            source,
323            reader.offset() as i64,
324            i64::from(num_weights) * 20,
325            surface_offset + SURFACE_SIZE,
326            end,
327            reader.offset() - 4,
328        )?;
329        let mut weights = Vec::with_capacity(num_weights as usize);
330        for _ in 0..num_weights {
331            weights.push(Md4Weight {
332                bone_index: md4_index(reader, source, num_bones, "weight bone")?,
333                bone_weight: reader.finite_f32()?,
334                offset: md4_vec3(reader)?,
335            });
336        }
337        vertices.push(Md4Vertex {
338            normal,
339            tex_coords,
340            weights,
341        });
342    }
343    Ok((
344        Md4Surface {
345            name,
346            shader,
347            vertices,
348            triangles,
349            bone_references,
350        },
351        end,
352    ))
353}

355/// Decode owned MD4 records (`parseMd4`).
356///
357/// Bytes after `ofsEnd` and unused bone-name metadata are ignored by the
358/// source loader.
359pub fn parse_md4(data: &[u8], source: &str) -> Result<DecodedMd4Model, BinaryError> {
360    let mut reader = BinaryReader::new(data, source);
361    md4_range(source, 0, HEADER_SIZE, 0, reader.length() as i64, 0)?;
362    if reader.u32()? != MD4_IDENT {
363        return Err(md4_fail(source, 0, "expected IDP4 model magic".to_string()));
364    }
365    let version = reader.i32()?;
366    if version != MD4_VERSION {
367        return Err(md4_fail(source, 4, format!("unsupported MD4 version {version}")));
368    }
369    let name = md4_string(&mut reader)?;
370    let num_frames = md4_count(&mut reader, source, i32::MAX)?;
371    if num_frames == 0 {
372        return Err(md4_fail(source, 72, "MD4 model has no frames".to_string()));
373    }
374    let num_bones = md4_count(&mut reader, source, 128)?;
375    let ofs_bone_names = reader.i32()?;
376    let ofs_frames = reader.i32()?;
377    let num_lods = md4_count(&mut reader, source, i32::MAX)?;
378    let ofs_lods = reader.i32()?;
379    let ofs_end = reader.i32()?;
380    md4_range(source, 0, i64::from(ofs_end), 0, reader.length() as i64, 96)?;
381    if ofs_end < HEADER_SIZE as i32 {
382        return Err(md4_fail(source, 96, "MD4 model end precedes its header".to_string()));
383    }
384    let frame_size = 40 + i64::from(num_bones) * 48;
385    md4_range(
386        source,
387        i64::from(ofs_frames),
388        i64::from(num_frames) * frame_size,
389        HEADER_SIZE,
390        i64::from(ofs_end),
391        84,
392    )?;
393    if num_lods > 0 {
394        md4_range(
395            source,
396            i64::from(ofs_lods),
397            i64::from(num_lods) * LOD_SIZE,
398            HEADER_SIZE,
399            i64::from(ofs_end),
400            92,
401        )?;
402    }
403
404    reader.seek(ofs_frames as usize)?;
405    let mut frames = Vec::with_capacity(num_frames as usize);
406    for _ in 0..num_frames {
407        let bounds = Bounds {
408            min: md4_vec3(&mut reader)?,
409            max: md4_vec3(&mut reader)?,
410        };
411        let local_origin = md4_vec3(&mut reader)?;
412        let radius = reader.finite_f32()?;
413        let mut bones = Vec::with_capacity(num_bones as usize);
414        for _ in 0..num_bones {
415            bones.push(Md4Bone {
416                matrix: [md4_row(&mut reader)?, md4_row(&mut reader)?, md4_row(&mut reader)?],
417            });
418        }
419        frames.push(Md4Frame {
420            bounds,
421            local_origin,
422            radius,
423            bones,
424        });
425    }
426
427    let mut lods = Vec::with_capacity(num_lods as usize);
428    let mut lod_offset = i64::from(ofs_lods);
429    for _ in 0..num_lods {
430        md4_range(
431            source,
432            lod_offset,
433            LOD_SIZE,
434            HEADER_SIZE,
435            i64::from(ofs_end),
436            lod_offset as usize,
437        )?;
438        reader.seek(lod_offset as usize)?;
439        let num_surfaces = md4_count(&mut reader, source, i32::MAX)?;
440        let ofs_surfaces = reader.i32()?;
441        let lod_length = reader.i32()?;
442        md4_range(
443            source,
444            lod_offset,
445            i64::from(lod_length),
446            lod_offset,
447            i64::from(ofs_end),
448            (lod_offset + 8) as usize,
449        )?;
450        if lod_length < LOD_SIZE as i32 {
451            return Err(md4_fail(
452                source,
453                (lod_offset + 8) as usize,
454                "MD4 LOD end precedes its header".to_string(),
455            ));
456        }
457        if num_surfaces > 0 {
458            md4_range(
459                source,
460                i64::from(ofs_surfaces),
461                i64::from(num_surfaces) * SURFACE_SIZE,
462                LOD_SIZE,
463                i64::from(lod_length),
464                (lod_offset + 4) as usize,
465            )?;
466        }
467        let mut surfaces = Vec::with_capacity(num_surfaces as usize);
468        let mut surface_offset = lod_offset + i64::from(ofs_surfaces);
469        for _ in 0..num_surfaces {
470            let (surface, next) = parse_surface(
471                &mut reader,
472                source,
473                surface_offset,
474                lod_offset + i64::from(lod_length),
475                num_bones,
476            )?;
477            surfaces.push(surface);
478            surface_offset = next;
479        }
480        lods.push(Md4Lod { surfaces });
481        lod_offset += i64::from(lod_length);
482    }
483    let mut bone_names = None;
484    // The renderer ignores this field. Decode names only when the unused table is present.
485    if i64::from(ofs_bone_names) >= HEADER_SIZE
486        && i64::from(ofs_bone_names) <= i64::from(ofs_end) - i64::from(num_bones) * 64
487    {
488        reader.seek(ofs_bone_names as usize)?;
489        let mut names = Vec::with_capacity(num_bones as usize);
490        for _ in 0..num_bones {
491            names.push(md4_string(&mut reader)?);
492        }
493        bone_names = Some(names);
494    }
495    let byte_length = ofs_end as usize;
496    Ok(DecodedMd4Model {
497        source: source.to_string(),
498        bytes: data[..byte_length].to_vec(),
499        bone_names,
500        model: Md4Model {
501            version,
502            name,
503            num_bones: num_bones as u32,
504            byte_length,
505            frames,
506            lods,
507        },
508    })
509}

511/// Skin one MD4 surface (`skinMd4Surface`).
512///
513/// `RB_SurfaceAnim` uses global bone indices, including when
514/// `bone_references` is sparse.
515///
516/// # Panics
517///
518/// Panics on invalid frame selections or short bone lists.
519#[must_use]
520pub fn skin_md4_surface(
521    model: &Md4Model,
522    surface: &Md4Surface,
523    frame: i32,
524    previous_frame: i32,
525    back_lerp: f32,
526) -> Vec<Md4SkinnedVertex> {
527    if frame < 0 || previous_frame < 0 || !back_lerp.is_finite() {
528        panic!("Invalid MD4 frame selection");
529    }
530    let current = at(&model.frames, frame as usize, "MD4 frame");
531    let back = if frame == previous_frame { 0.0 } else { back_lerp };
532    let old = if back == 0.0 {
533        current
534    } else {
535        at(&model.frames, previous_frame as usize, "MD4 previous frame")
536    };
537    let front = 1.0 - back;
538    let blend = |a: f32, b: f32| if back == 0.0 { a } else { front * a + back * b };
539    let bones: Vec<Md4Bone> = current
540        .bones
541        .iter()
542        .enumerate()
543        .map(|(index, bone)| {
544            let before = at(&old.bones, index, "MD4 bone");
545            let row = |a: Vec4, b: Vec4| vec4(blend(a.x, b.x), blend(a.y, b.y), blend(a.z, b.z), blend(a.w, b.w));
546            Md4Bone {
547                matrix: [
548                    row(bone.matrix[0], before.matrix[0]),
549                    row(bone.matrix[1], before.matrix[1]),
550                    row(bone.matrix[2], before.matrix[2]),
551                ],
552            }
553        })
554        .collect();
555    let dot = |row: Vec4, value: Vec3| dot3(vec3(row.x, row.y, row.z), value);
556    surface
557        .vertices
558        .iter()
559        .map(|vertex| {
560            let (mut x, mut y, mut z) = (0.0f32, 0.0f32, 0.0f32);
561            let (mut nx, mut ny, mut nz) = (0.0f32, 0.0f32, 0.0f32);
562            for weight in &vertex.weights {
563                let bone = at(&bones, weight.bone_index as usize, "MD4 weight bone");
564                let (a, b, c) = (bone.matrix[0], bone.matrix[1], bone.matrix[2]);
565                x += weight.bone_weight * (dot(a, weight.offset) + a.w);
566                y += weight.bone_weight * (dot(b, weight.offset) + b.w);
567                z += weight.bone_weight * (dot(c, weight.offset) + c.w);
568                nx += weight.bone_weight * dot(a, vertex.normal);
569                ny += weight.bone_weight * dot(b, vertex.normal);
570                nz += weight.bone_weight * dot(c, vertex.normal);
571            }
572            Md4SkinnedVertex {
573                position: vec3(x, y, z),
574                normal: vec3(nx, ny, nz),
575            }
576        })
577        .collect()
578}
